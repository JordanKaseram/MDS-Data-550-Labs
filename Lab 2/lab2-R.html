free_genres <- movies %>%
  unnest(genres)

genre_meds <- free_genres %>%
  group_by(genres) %>%
  summarise(
    median_revenue = median(revenue, na.rm = TRUE),
    .groups = "drop"
  )

ggplot(free_genres, aes(x = revenue, y = genres)) +
  geom_violin(
    trim = FALSE,
    fill = "blue",
    alpha = 0.7
  ) +
  geom_point(
    data = genre_meds,
    aes(x = median_revenue, y = genres),
    color = "green",
    size = 2
  ) +
  scale_x_continuous(labels = scales::comma) +
  labs(
    title = "Revenue Distribution by Genre",
    x = "Revenue (USD)",
    y = "Genre"
  ) +
  theme_linedraw()

free_genres <- movies %>%
  unnest(genres)

genre_order <- free_genres %>%
  group_by(genres) %>%
  summarise(median_revenue = median(revenue, na.rm = TRUE)) %>%
  arrange(median_revenue) %>%
  pull(genres)

free_genres <- free_genres %>%
  mutate(genres = factor(genres, levels = genre_order))

movies_long <- free_genres %>%
  pivot_longer(
    cols = where(is.numeric),
    names_to = "variable",
    values_to = "value"
  )

ggplot(movies_long, aes(x = value, y = genres)) +
  geom_violin(
    trim = FALSE,
    fill = "blue",
    alpha = 0.7
  ) +
  facet_wrap(~ variable, scales = "free_x") +
  scale_x_continuous(labels = scales::comma) +
  labs(
    title = "Distributions of Numerical Variables by Genre",
    x = "Value",
    y = "Genre"
  ) +
  theme_linedraw()
